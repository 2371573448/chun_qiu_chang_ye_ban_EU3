country_event = {
	id = 8754
	is_triggered_only = yes
	title = "EVTNAME8754"
	desc = "EVTDESC8754"
	option = {
		name = "EVTOPTA8754"
		random_list = {
			3333 = {
				set_country_flag = suijiqi_1
			}
			3333 = {
				set_country_flag = suijiqi_2
			}
			3333 = {
				set_country_flag = suijiqi_3
			}
		}
		any_country = { 
			limit = { 
				tag = NPC
				has_country_flag = suijiqi_1_shou
				has_country_flag = suijiqi_2
				has_country_flag = suijiqi_3
			}
			any_country = { 
				limit = { 
					has_country_flag = suijiqi_mubiao
				}
				set_country_flag = suijiqi_jieguo_1
			}
		}
		any_country = { 
			limit = { 
				tag = NPC
				has_country_flag = suijiqi_1
				has_country_flag = suijiqi_2_shou
				has_country_flag = suijiqi_3
			}
			any_country = { 
				limit = { 
					has_country_flag = suijiqi_mubiao
				}
				set_country_flag = suijiqi_jieguo_2
			}
		}
		any_country = { 
			limit = { 
				tag = NPC
				has_country_flag = suijiqi_1
				has_country_flag = suijiqi_2
				has_country_flag = suijiqi_3_shou
			}
			any_country = { 
				limit = { 
					has_country_flag = suijiqi_mubiao
				}
				set_country_flag = suijiqi_jieguo_3
			}
		}
		any_country = { 
			limit = { 
				tag = NPC
				not = { 
					or = {
						and = {
							has_country_flag = suijiqi_1_shou
							has_country_flag = suijiqi_2
							has_country_flag = suijiqi_3
						}
						and = {
							has_country_flag = suijiqi_1
							has_country_flag = suijiqi_2_shou
							has_country_flag = suijiqi_3
						}
						and = {
							has_country_flag = suijiqi_1
							has_country_flag = suijiqi_2
							has_country_flag = suijiqi_3_shou
						}
					}
				}
			}
			country_event = 8752
		}
	}
}
